CREATE TABLE public.users (
    user_id SERIAL PRIMARY KEY,
    fio VARCHAR(255) NOT NULL,
    weight NUMERIC(5,2) NOT NULL,
    height NUMERIC(5,2) NOT NULL,
    goal VARCHAR(255) NOT NULL
);

CREATE TABLE public.exercise_group (
    group_id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    norm_calories INT NOT NULL,
    load_type VARCHAR(100) NOT NULL
);

CREATE TABLE public.record (
    record_id SERIAL PRIMARY KEY,
    user_id INT NOT NULL,
    group_id INT NOT NULL,
    calories INT NOT NULL,
    record_date DATE NOT NULL,
    CONSTRAINT fk_record_user FOREIGN KEY (user_id) REFERENCES public.users(user_id) ON DELETE CASCADE,
    CONSTRAINT fk_record_group FOREIGN KEY (group_id) REFERENCES public.exercise_group(group_id) ON DELETE CASCADE
);
